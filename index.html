Configurar sincronização ✦

Pra sua estante funcionar no celular e no computador com os mesmos dados, configure o Supabase. É gratuito e leva 3 minutos.

Passo a passo:
1. Cria conta grátis em supabase.com
2. Clica em "New project" (qualquer nome, senha e região)
3. Espera ~2 min até ficar pronto
4. Vai em SQL Editor e cola o SQL abaixo
5. Roda ▶ (Run)
6. Depois pega em Settings → API:
   • Project URL
   • anon public key
7. Cola aqui embaixo ✦
-- cole no SQL Editor e rode -- SE JÁ CRIOU O BANCO ANTES, rode só isso: -- ALTER TABLE books ADD COLUMN IF NOT EXISTS user_id uuid REFERENCES auth.users(id); -- UPDATE books SET user_id = auth.uid() WHERE user_id IS NULL; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS fav_rank int; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS top_general boolean DEFAULT false; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS top_general_rank int; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS series_type text; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS series_name text; -- ALTER TABLE books ADD COLUMN IF NOT EXISTS series_volume int; -- PARA BANCO NOVO, rode tudo abaixo: create table books ( id bigint generated by default as identity primary key, user_id uuid references auth.users(id), title text not null, author text not null, genre text not null, status text not null default 'pendente', start_date date, end_date date, rating int, favorite boolean default false, pages_total int, pages_current int default 0, fav_rank int, top_general boolean default false, top_general_rank int, series_type text, series_name text, series_volume int, created_at timestamptz default now() ); create table reading_sessions ( id bigint generated by default as identity primary key, book_id bigint not null references books(id) on delete cascade, started_at timestamptz not null, ended_at timestamptz not null, duration_seconds int not null, page_start int, page_end int, created_at timestamptz default now() ); create index idx_sessions_book on reading_sessions(book_id); create index idx_books_user on books(user_id); create table genres ( id text primary key, name text not null, emoji text, color text, preview text, palette text[], sort_order int default 0, created_at timestamptz default now() ); alter table books enable row level security; alter table reading_sessions enable row level security; alter table genres enable row level security; -- Grants necessários para o app acessar as tabelas pela anon key/JWT grant usage on schema public to anon, authenticated; grant all on table books to authenticated; grant all on table reading_sessions to authenticated; grant select on table genres to anon, authenticated; grant usage, select on all sequences in schema public to authenticated; -- Política: cada usuário vê só seus livros create policy "usuario ve seus livros" on books for all using (auth.uid() = user_id) with check (auth.uid() = user_id); -- Sessões: acessíveis via livro do usuário create policy "usuario ve suas sessoes" on reading_sessions for all using ( exists (select 1 from books where books.id = reading_sessions.book_id and books.user_id = auth.uid()) ) with check ( exists (select 1 from books where books.id = reading_sessions.book_id and books.user_id = auth.uid()) ); create policy "acesso total genres" on genres for all using (true) with check (true); -- WISHLIST (séries em lançamento) create table if not exists wishlist ( id bigint generated by default as identity primary key, user_id uuid references auth.users(id), title text not null, author text not null, genre text not null, total_volumes int, released_volumes int default 0, created_at timestamptz default now() ); alter table wishlist enable row level security; grant all on table wishlist to authenticated; grant usage, select on all sequences in schema public to authenticated; create policy "usuario ve sua wishlist" on wishlist for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
